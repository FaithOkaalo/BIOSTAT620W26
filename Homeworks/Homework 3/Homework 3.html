Question 3 – Tokenize the abstracts into bigrams.10 most common bigrams and visualize them.

bigrams <- pubmed %>%
  unnest_tokens(bigram, abstract, token = "ngrams", n = 2)
bigrams_sep <- bigrams %>%
  separate(bigram, into = c("word1", "word2"), sep = " ")
bigrams_filtered <- bigrams_sep %>%
  filter(!word1 %in% stop_words$word,
         !word2 %in% stop_words$word,
         !str_detect(word1, "^\\d+$"),
         !str_detect(word2, "^\\d+$"),
         nchar(word1) > 2, nchar(word2) > 2)
top10_bigrams <- bigrams_filtered %>%
  unite(bigram, word1, word2, sep = " ") %>%   
  count(bigram, sort = TRUE) %>%
  slice_max(n, n = 10)

Question 6: Sentiment analysis with AFINN lexicon to get an average positivity score for each abstract

afinn <- get_sentiments("afinn")

# Add an abstract index so we can group by individual abstract
pubmed_indexed <- pubmed %>%
  mutate(abstract_id = row_number())   

# Tokenize, join AFINN, sum score per abstract
afinn_scores <- pubmed_indexed %>%
  unnest_tokens(word, abstract) %>%
  inner_join(afinn, by = "word") %>%           
  group_by(abstract_id, term) %>%
  summarise(score = sum(value), .groups = "drop") 

# Summary stats by term
afinn_scores %>%
  group_by(term) %>%
  summarise(mean = mean(score), median = median(score),
            sd = sd(score), min = min(score), max = max(score))
# A tibble: 5 × 6
  term              mean median    sd   min   max

Most search terms had negative average sentiment scores. Prostate cancer had the most negative score (mean -2.64), followed by COVID (mean -2.29). This may indicate that abstracts related to these topics had a more negative language. Preeclampsia and meningitis also had negative average sentiment, though less extreme. On the contrary, cystic fibrosis stands out as noticeably different, with a positive average sentiment score (mean 2.03), suggesting more positive or neutral language in those abstracts.
